document.addEventListener("DOMContentLoaded", () => { // Initialize the engine UI load sequence fetchKingVMetrics(); }); /** * Async fetch engine to ingest calculations from the local GRC engine */ async function fetchKingVMetrics() { const endpoint = "/api/v1/compliance/king-v/metrics"; try { const response = await fetch(endpoint); if (!response.ok) { throw new Error(`HTTP Error! Status: ${response.status}`); } const data = await response.json(); // Render top-level core KPI matrices updateGlobalKPIs(data.overall_compliance_score, data.trust_dividend_index); // Render structural functional groups renderGoverningFunctions(data.governing_functions); } catch (error) { console.error("Failed to render engine analytics dashboard:", error); showDashboardErrorState(error.message); } } /** * Updates high-level scoring metrics on the dashboard header cards */ function updateGlobalKPIs(complianceScore, trustDividend) { const complianceEl = document.getElementById("overall-compliance-score"); const trustEl = document.getElementById("trust-dividend-index"); if (complianceEl) complianceEl.textContent = `${complianceScore}%`; if (trustEl) trustEl.textContent = `${trustDividend}%`; } /** * Iterates through the 4 core governing functions of King V and injects principles */ function renderGoverningFunctions(functionsData) { const container = document.getElementById("king-v-matrix-container"); if (!container) return; container.innerHTML = ""; // Clear existing skeletons or placeholder data // Human-readable titles mapped to API schema dictionary keys const categoryTitles = { steering_direction: "Steering & Setting Direction", policy_planning: "Policy & Planning", oversight_monitoring: "Oversight & Monitoring", accountability: "Accountability & Disclosure Framework" }; for (const [key, category] of Object.entries(functionsData)) { const sectionTitle = categoryTitles[key] || key.replace("_", " "); // Generate functional block element wrapper const sectionHTML = `

${sectionTitle}

Category Score: ${category.category_score}%
${category.principles.map(p => generatePrincipleCard(p)).join('')}
`; container.insertAdjacentHTML('beforeend', sectionHTML); } } /** * Factory component generating individual UI matrix elements for the 13 principles */ function generatePrincipleCard(principle) { // Algorithmic color determination for compliance thresholds let badgeColor = "#e53e3e"; // Non-Compliant Red let badgeBg = "#fff5f5"; if (principle.status === "Compliant") { badgeColor = "#38a169"; // Compliant Green badgeBg = "#f0fff4"; } else if (principle.status === "In Progress") { badgeColor = "#dd6b20"; // In Progress Orange badgeBg = "#fffaf0"; } return `
${principle.principle_id} ${principle.status}

${principle.title}

Engine Score: ${principle.score}%
Gates Validated: ${principle.gates_passed} / ${principle.gates_total}
`; } /** * Renders UI notification feedback if backend validation fails */ function showDashboardErrorState(errorMessage) { const container = document.getElementById("king-v-matrix-container"); if (container) { container.innerHTML = `
Local Engine Error: Failed to fetch compliance matrix records. Line diagnostic: ${errorMessage}
`; } }
